import os import logging from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup from telegram.ext import ApplicationBuilder, CommandHandler, CallbackQueryHandler, ContextTypes from telegram.constants import ChatMemberStatus logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s") log = logging.getLogger("itachi-bot") BOT_TOKEN = os.environ["BOT_TOKEN"] CHANNEL = os.environ["CHANNEL"] # "@your_channel" ya "-1001234567890" WEB_URL = os.environ["WEB_URL"] # https://itachi-panel.onrender.com/index.html WELCOME = ( "🔥 ITACHI · TSUKUYOMI\n\n" "Panel use karne ke liye pehle channel join karo.\n" "Join ke baad niche button dabao." ) def join_kb(): ch = CHANNEL.lstrip("@") return InlineKeyboardMarkup([ [InlineKeyboardButton("📢 Join Channel", url=f"https://t.me/{ch}")], [InlineKeyboardButton("✅ Verify & Open Panel", callback_data="verify")], ]) def open_kb(): return InlineKeyboardMarkup([ [InlineKeyboardButton("🔥 Open Panel", url=WEB_URL)], ]) async def is_member(ctx: ContextTypes.DEFAULT_TYPE, user_id: int) -> bool: try: m = await ctx.bot.get_chat_member(chat_id=CHANNEL, user_id=user_id) return m.status in ( ChatMemberStatus.MEMBER, ChatMemberStatus.ADMINISTRATOR, ChatMemberStatus.OWNER, ChatMemberStatus.RESTRICTED, ) except Exception as e: log.warning("member check failed: %s", e) return False async def start(update: Update, ctx: ContextTypes.DEFAULT_TYPE): uid = update.effective_user.id if await is_member(ctx, uid): await update.message.reply_text( "✅ Verified. Panel ready hai.", reply_markup=open_kb(), ) else: await update.message.reply_text(WELCOME, reply_markup=join_kb()) async def verify(update: Update, ctx: ContextTypes.DEFAULT_TYPE): q = update.callback_query await q.answer() uid = q.from_user.id if await is_member(ctx, uid): await q.edit_message_text( "✅ Verified. Panel ready hai.", reply_markup=open_kb(), ) else: await q.answer("Pehle channel join karo.", show_alert=True) try: await q.edit_message_reply_markup(reply_markup=join_kb()) except Exception: pass async def gate_all(update: Update, ctx: ContextTypes.DEFAULT_TYPE): uid = update.effective_user.id if await is_member(ctx, uid): await update.message.reply_text( "✅ Verified. Panel ready hai.", reply_markup=open_kb(), ) else: await update.message.reply_text(WELCOME, reply_markup=join_kb()) def main(): app = ApplicationBuilder().token(BOT_TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(CallbackQueryHandler(verify, pattern="^verify$")) # har text message pe gate from telegram.ext import MessageHandler, filters app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, gate_all)) log.info("bot up") app.run_polling() if __name__ == "__main__": main()
